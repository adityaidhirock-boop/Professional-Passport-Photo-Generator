class HumanSegmentationDataset(Dataset):
    def __init__(self, images_dir, masks_dir, transform=None, img_size=(320, 320)):
        self.images_dir = images_dir
        self.masks_dir = masks_dir
        self.transform = transform
        self.img_size = img_size

        # Get all image files
        self.image_files = [f for f in os.listdir(images_dir)
                           if f.lower().endswith(('.png', '.jpg', '.jpeg'))]

        print(f" Dataset initialized with {len(self.image_files)} images")

    def __len__(self):
        return len(self.image_files)

    def __getitem__(self, idx):
        # Load image
        img_name = self.image_files[idx]
        img_path = os.path.join(self.images_dir, img_name)
        image = cv2.imread(img_path, cv2.IMREAD_COLOR)
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)

        # Load mask - handle different naming conventions
        mask_name = img_name.replace('.jpg', '.png').replace('.jpeg', '.png')
        if not mask_name.endswith('.png'):
            mask_name = os.path.splitext(img_name)[0] + '.png'

        mask_path = os.path.join(self.masks_dir, mask_name)

        if not os.path.exists(mask_path):
            # Try alternative naming
            mask_name = os.path.splitext(img_name)[0] + '_mask.png'
            mask_path = os.path.join(self.masks_dir, mask_name)

        if os.path.exists(mask_path):
            mask = cv2.imread(mask_path, cv2.IMREAD_GRAYSCALE)
        else:
            print(f" Mask not found for {img_name}, creating empty mask")
            mask = np.zeros((image.shape[0], image.shape[1]), dtype=np.uint8)

        # Resize
        image = cv2.resize(image, self.img_size)
        mask = cv2.resize(mask, self.img_size)

        # Normalize mask to 0-1
        mask = (mask > 128).astype(np.float32)

        # Apply augmentations
        if self.transform:
            augmented = self.transform(image=image, mask=mask)
            image = augmented['image']
            mask = augmented['mask']

        return image, mask.unsqueeze(0)  # Add channel dimension to mask

class EarlyStopping:
    def __init__(self, patience=7, min_delta=0, restore_best_weights=True):
        self.patience = patience
        self.min_delta = min_delta
        self.restore_best_weights = restore_best_weights
        self.best_loss = None
        self.counter = 0
        self.early_stop = False
        self.best_weights = None

    def __call__(self, val_loss, model):
        if self.best_loss is None:
            self.best_loss = val_loss
            if self.restore_best_weights:
                self.best_weights = model.state_dict().copy()
        elif val_loss < self.best_loss - self.min_delta:
            self.best_loss = val_loss
            self.counter = 0
            if self.restore_best_weights:
                self.best_weights = model.state_dict().copy()
        else:
            self.counter += 1
            if self.counter >= self.patience:
                self.early_stop = True
                if self.restore_best_weights and self.best_weights:
                    model.load_state_dict(self.best_weights)

class ResNetEncoder(nn.Module):
    """Pretrained ResNet encoder for U2-Net"""
    def __init__(self, encoder_name='resnet34', pretrained=True):
        super().__init__()

        if encoder_name == 'resnet34':
            resnet = models.resnet34(pretrained=pretrained)
            self.channels = [64, 64, 128, 256, 512]
        elif encoder_name == 'resnet50':
            resnet = models.resnet50(pretrained=pretrained)
            self.channels = [64, 256, 512, 1024, 2048]
        else:
            raise ValueError(f"Unsupported encoder: {encoder_name}")

        # Extract ResNet layers
        self.conv1 = resnet.conv1
        self.bn1 = resnet.bn1
        self.relu = resnet.relu
        self.maxpool = resnet.maxpool

        self.layer1 = resnet.layer1
        self.layer2 = resnet.layer2
        self.layer3 = resnet.layer3
        self.layer4 = resnet.layer4

        print(f"Loaded pretrained {encoder_name} encoder from ImageNet")

    def forward(self, x):
        features = []

        # Stage 0
        x = self.conv1(x)
        x = self.bn1(x)
        x = self.relu(x)
        features.append(x)

        # Stage 1
        x = self.maxpool(x)
        x = self.layer1(x)
        features.append(x)

        # Stage 2
        x = self.layer2(x)
        features.append(x)

        # Stage 3
        x = self.layer3(x)
        features.append(x)

        # Stage 4
        x = self.layer4(x)
        features.append(x)

        return features

class HybridU2Net(nn.Module):
    """Completely Fixed Hybrid U2-Net with proper channel flow"""

    def __init__(self, encoder_name='resnet34', pretrained=True, num_classes=1):
        super().__init__()

        # Pretrained ResNet encoder
        self.encoder = ResNetEncoder(encoder_name, pretrained)
        encoder_channels = self.encoder.channels  # [64, 64, 128, 256, 512]

        print(f"Encoder channels: {encoder_channels}")

        # Simplified decoder with proper channel flow
        # Decoder path: 512 -> 256 -> 128 -> 64 -> 32

        # Stage 1: 512 -> 256 (from deepest encoder)
        self.dec4 = nn.Sequential(
            nn.ConvTranspose2d(512, 256, 2, stride=2),
            nn.BatchNorm2d(256),
            nn.ReLU(inplace=True),
            nn.Conv2d(256, 256, 3, padding=1),
            nn.BatchNorm2d(256),
            nn.ReLU(inplace=True)
        )

        # Stage 2: 256 + 256 (skip) -> 128
        self.dec3 = nn.Sequential(
            nn.ConvTranspose2d(512, 128, 2, stride=2),  # 256+256=512 input
            nn.BatchNorm2d(128),
            nn.ReLU(inplace=True),
            nn.Conv2d(128, 128, 3, padding=1),
            nn.BatchNorm2d(128),
            nn.ReLU(inplace=True)
        )

        # Stage 3: 128 + 128 (skip) -> 64
        self.dec2 = nn.Sequential(
            nn.ConvTranspose2d(256, 64, 2, stride=2),   # 128+128=256 input
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),
            nn.Conv2d(64, 64, 3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True)
        )

        # Stage 4: 64 + 64 (skip) -> 32
        self.dec1 = nn.Sequential(
            nn.ConvTranspose2d(128, 32, 2, stride=2),   # 64+64=128 input
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
            nn.Conv2d(32, 32, 3, padding=1),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True)
        )

        # Final output: 32 + 64 (first encoder) -> 1
        self.final_conv = nn.Sequential(
            nn.Conv2d(96, 32, 3, padding=1),  # 32+64=96 input
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
            nn.Conv2d(32, num_classes, 1),
            nn.Sigmoid()
        )

        # Side outputs for deep supervision
        self.side4 = nn.Conv2d(256, num_classes, 1)
        self.side3 = nn.Conv2d(128, num_classes, 1)
        self.side2 = nn.Conv2d(64, num_classes, 1)
        self.side1 = nn.Conv2d(32, num_classes, 1)

        print("Fixed architecture initialized successfully")

    def forward(self, x):
        # Get input size for final resizing
        input_size = x.shape[2:]

        # Encoder forward pass
        enc_features = self.encoder(x)
        # enc_features[0]: 64 channels, H/4 x W/4
        # enc_features[1]: 64 channels, H/4 x W/4
        # enc_features[2]: 128 channels, H/8 x W/8
        # enc_features[3]: 256 channels, H/16 x W/16
        # enc_features[4]: 512 channels, H/32 x W/32

        # Decoder Stage 1: 512 -> 256
        dec4 = self.dec4(enc_features[4])  # 512 -> 256, size becomes H/16 x W/16

        # Skip connection with enc_features[3] (256 channels, H/16 x W/16)
        dec4_skip = torch.cat([dec4, enc_features[3]], dim=1)  # 256+256=512 channels

        # Decoder Stage 2: 512 -> 128
        dec3 = self.dec3(dec4_skip)  # 512 -> 128, size becomes H/8 x W/8

        # Skip connection with enc_features[2] (128 channels, H/8 x W/8)
        dec3_skip = torch.cat([dec3, enc_features[2]], dim=1)  # 128+128=256 channels

        # Decoder Stage 3: 256 -> 64
        dec2 = self.dec2(dec3_skip)  # 256 -> 64, size becomes H/4 x W/4

        # Skip connection with enc_features[1] (64 channels, H/4 x W/4)
        dec2_skip = torch.cat([dec2, enc_features[1]], dim=1)  # 64+64=128 channels

        # Decoder Stage 4: 128 -> 32
        dec1 = self.dec1(dec2_skip)  # 128 -> 32, size becomes H/2 x W/2

        # Upsample to match first encoder feature size
        dec1_up = F.interpolate(dec1, size=enc_features[0].shape[2:], mode='bilinear', align_corners=False)

        # Skip connection with enc_features[0] (64 channels)
        final_skip = torch.cat([dec1_up, enc_features[0]], dim=1)  # 32+64=96 channels

        # Final output
        main_output = self.final_conv(final_skip)  # 96 -> 1

        # Resize main output to input size
        main_output = F.interpolate(main_output, size=input_size, mode='bilinear', align_corners=False)

        # Side outputs
        side4 = torch.sigmoid(self.side4(dec4))
        side4 = F.interpolate(side4, size=input_size, mode='bilinear', align_corners=False)

        side3 = torch.sigmoid(self.side3(dec3))
        side3 = F.interpolate(side3, size=input_size, mode='bilinear', align_corners=False)

        side2 = torch.sigmoid(self.side2(dec2))
        side2 = F.interpolate(side2, size=input_size, mode='bilinear', align_corners=False)

        side1 = torch.sigmoid(self.side1(dec1))
        side1 = F.interpolate(side1, size=input_size, mode='bilinear', align_corners=False)

        side_outputs = [side4, side3, side2, side1]

        return main_output, side_outputs

class HybridLoss(nn.Module):
    """Combined loss for hybrid model with multi-scale supervision"""

    def __init__(self, main_weight=1.0, side_weight=0.4, bce_weight=0.3, dice_weight=0.7):
        super().__init__()
        self.main_weight = main_weight
        self.side_weight = side_weight
        self.bce_weight = bce_weight
        self.dice_weight = dice_weight
        self.bce = nn.BCELoss()

    def dice_loss(self, pred, target, smooth=1.0):
        pred_flat = pred.view(-1)
        target_flat = target.view(-1)
        intersection = (pred_flat * target_flat).sum()
        return 1 - (2.0 * intersection + smooth) / (pred_flat.sum() + target_flat.sum() + smooth)

    def combined_loss(self, pred, target):
        bce = self.bce(pred, target)
        dice = self.dice_loss(pred, target)
        return self.bce_weight * bce + self.dice_weight * dice

    def forward(self, outputs, target):
        main_output, side_outputs = outputs

        # Main output loss
        main_loss = self.combined_loss(main_output, target)

        # Side outputs loss
        side_loss = 0
        for side_out in side_outputs:
            side_loss += self.combined_loss(side_out, target)

        total_loss = self.main_weight * main_loss + self.side_weight * side_loss
        return total_loss, main_loss.item(), side_loss.item()

def calculate_iou(pred, target, threshold=0.5):
    """Calculate IoU for validation"""
    pred_binary = (pred > threshold).float()
    target_binary = (target > threshold).float()

    intersection = (pred_binary * target_binary).sum()
    union = pred_binary.sum() + target_binary.sum() - intersection

    if union == 0:
        return 1.0 if intersection == 0 else 0.0

    return (intersection / union).item()

def calculate_dice_score(pred, target, threshold=0.5):
    """Calculate Dice score"""
    pred_binary = (pred > threshold).float()
    target_binary = (target > threshold).float()

    intersection = (pred_binary * target_binary).sum()
    dice = (2.0 * intersection) / (pred_binary.sum() + target_binary.sum())

    return dice.item() if not torch.isnan(dice) else 0.0

class PassportDataset(Dataset):
    def __init__(self, image_dir, mask_dir, image_size=320, transform=None):
        self.image_dir = image_dir
        self.mask_dir = mask_dir
        self.image_size = image_size
        self.transform = transform

        self.image_files = []
        self.mask_files = []

        self._load_dataset()

    def _load_dataset(self):
        if not os.path.exists(self.image_dir):
            print(f"Image directory not found: {self.image_dir}")
            return

        if not os.path.exists(self.mask_dir):
            print(f"Mask directory not found: {self.mask_dir}")
            return

        image_extensions = ['.jpg', '.jpeg', '.png', '.bmp']
        all_images = []

        for ext in image_extensions:
            all_images.extend([f for f in os.listdir(self.image_dir) if f.lower().endswith(ext)])

        for img_file in all_images:
            img_path = os.path.join(self.image_dir, img_file)

            base_name = os.path.splitext(img_file)[0]
            mask_candidates = [
                f"{base_name}.png",
                f"{base_name}.jpg",
                f"{base_name}_mask.png",
                f"{base_name}_gt.png"
            ]

            mask_path = None
            for mask_candidate in mask_candidates:
                candidate_path = os.path.join(self.mask_dir, mask_candidate)
                if os.path.exists(candidate_path):
                    mask_path = candidate_path
                    break

            if mask_path:
                self.image_files.append(img_path)
                self.mask_files.append(mask_path)

        print(f"Dataset loaded: {len(self.image_files)} image-mask pairs")

    def __len__(self):
        return len(self.image_files)

    def __getitem__(self, idx):
        image_path = self.image_files[idx]
        mask_path = self.mask_files[idx]

        # Load image
        image = cv2.imread(image_path)
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        image = cv2.resize(image, (self.image_size, self.image_size))

        # Load mask
        mask = cv2.imread(mask_path, cv2.IMREAD_GRAYSCALE)
        mask = cv2.resize(mask, (self.image_size, self.image_size))

        # Normalize image
        image = image.astype(np.float32) / 255.0

        # Binarize mask
        mask = (mask > 127).astype(np.float32)

        # Convert to tensors
        image = torch.from_numpy(image).permute(2, 0, 1)
        mask = torch.from_numpy(mask).unsqueeze(0)

        # Apply transforms if provided
        if self.transform:
            image = self.transform(image)

        return image, mask

class TrainingEngine:
    def __init__(self, model, criterion, optimizer, scheduler, device, checkpoint_dir):
        self.model = model
        self.criterion = criterion
        self.optimizer = optimizer
        self.scheduler = scheduler
        self.device = device
        self.checkpoint_dir = checkpoint_dir

        self.training_history = {
            'train_loss': [],
            'val_loss': [],
            'val_iou': [],
            'val_dice': [],
            'learning_rates': []
        }

        self.best_iou = 0.0
        self.current_epoch = 0

        # Ensure checkpoint directory exists
        os.makedirs(checkpoint_dir, exist_ok=True)

    def calculate_iou(self, pred, target, threshold=0.5):
        pred_binary = (pred > threshold).float()
        target_binary = target

        intersection = (pred_binary * target_binary).sum()
        union = pred_binary.sum() + target_binary.sum() - intersection

        if union == 0:
            return 1.0 if intersection == 0 else 0.0

        return (intersection / union).item()

    def calculate_dice_score(self, pred, target, threshold=0.5):
        pred_binary = (pred > threshold).float()
        target_binary = target

        intersection = (pred_binary * target_binary).sum()
        dice = (2.0 * intersection) / (pred_binary.sum() + target_binary.sum() + 1e-8)

        return dice.item()

    def train_epoch(self, train_loader):
        self.model.train()
        total_loss = 0.0
        num_batches = len(train_loader)

        progress_bar = tqdm(train_loader, desc=f"Training Epoch {self.current_epoch + 1}")

        for batch_idx, (images, masks) in enumerate(progress_bar):
            images = images.to(self.device)
            masks = masks.to(self.device)

            self.optimizer.zero_grad()

            # Forward pass
            outputs = self.model(images)
            loss, main_loss, side_loss = self.criterion(outputs, masks)

            # Backward pass
            loss.backward()
            self.optimizer.step()

            batch_loss = loss.item()
            total_loss += batch_loss

            # Update progress bar
            progress_bar.set_postfix({
                'Loss': f'{batch_loss:.4f}',
                'Avg Loss': f'{total_loss/(batch_idx+1):.4f}'
            })

        return total_loss / num_batches

    def validate_epoch(self, val_loader):
        self.model.eval()
        total_loss = 0.0
        total_iou = 0.0
        total_dice = 0.0
        num_batches = len(val_loader)

        progress_bar = tqdm(val_loader, desc="Validation")

        with torch.no_grad():
            for images, masks in progress_bar:
                images = images.to(self.device)
                masks = masks.to(self.device)

                # Forward pass
                outputs = self.model(images)
                loss, _, _ = self.criterion(outputs, masks)

                # Get main output
                main_output = outputs[0] if isinstance(outputs, tuple) else outputs

                # Calculate metrics
                batch_loss = loss.item()
                batch_iou = self.calculate_iou(main_output, masks)
                batch_dice = self.calculate_dice_score(main_output, masks)

                total_loss += batch_loss
                total_iou += batch_iou
                total_dice += batch_dice

                # Update progress bar
                progress_bar.set_postfix({
                    'Loss': f'{batch_loss:.4f}',
                    'IoU': f'{batch_iou:.4f}',
                    'Dice': f'{batch_dice:.4f}'
                })

        avg_loss = total_loss / num_batches
        avg_iou = total_iou / num_batches
        avg_dice = total_dice / num_batches

        return avg_loss, avg_iou, avg_dice

    def save_checkpoint(self, epoch, is_best=False):
        checkpoint = {
            'epoch': epoch,
            'model_state_dict': self.model.state_dict(),
            'optimizer_state_dict': self.optimizer.state_dict(),
            'scheduler_state_dict': self.scheduler.state_dict(),
            'best_iou': self.best_iou,
            'training_history': self.training_history
        }

        # Save regular checkpoint every epoch
        checkpoint_path = os.path.join(self.checkpoint_dir, f'checkpoint_epoch_{epoch+1:03d}.pth')
        torch.save(checkpoint, checkpoint_path)
        logging.info(f"Checkpoint saved: {checkpoint_path}")

        # Save best model
        if is_best:
            best_path = os.path.join(self.checkpoint_dir, 'best_model.pth')
            torch.save(checkpoint, best_path)
            logging.info(f"Best model saved: {best_path}")

        return checkpoint_path

    def load_checkpoint(self, checkpoint_path):
        checkpoint = torch.load(checkpoint_path, map_location=self.device)

        self.model.load_state_dict(checkpoint['model_state_dict'])
        self.optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
        self.scheduler.load_state_dict(checkpoint['scheduler_state_dict'])

        self.current_epoch = checkpoint['epoch'] + 1
        self.best_iou = checkpoint['best_iou']
        self.training_history = checkpoint['training_history']

        logging.info(f"Checkpoint loaded from epoch {checkpoint['epoch']}")
        logging.info(f"Best IoU: {self.best_iou:.4f}")

    def train(self, train_loader, val_loader, num_epochs, early_stopping_patience=15):
        logging.info(f"Starting training for {num_epochs} epochs")
        logging.info(f"Device: {self.device}")
        logging.info(f"Early stopping patience: {early_stopping_patience}")

        start_time = time.time()
        epochs_without_improvement = 0

        for epoch in range(self.current_epoch, num_epochs):
            self.current_epoch = epoch

            logging.info(f"\nEpoch {epoch + 1}/{num_epochs}")
            logging.info("-" * 50)

            # Training phase
            train_loss = self.train_epoch(train_loader)

            # Validation phase
            val_loss, val_iou, val_dice = self.validate_epoch(val_loader)

            # Update learning rate
            self.scheduler.step(val_loss)
            current_lr = self.optimizer.param_groups[0]['lr']

            # Save training history
            self.training_history['train_loss'].append(train_loss)
            self.training_history['val_loss'].append(val_loss)
            self.training_history['val_iou'].append(val_iou)
            self.training_history['val_dice'].append(val_dice)
            self.training_history['learning_rates'].append(current_lr)

            # Log epoch results
            logging.info(f"Train Loss: {train_loss:.6f}")
            logging.info(f"Val Loss:   {val_loss:.6f}")
            logging.info(f"Val IoU:    {val_iou:.4f} ({val_iou*100:.1f}%)")
            logging.info(f"Val Dice:   {val_dice:.4f} ({val_dice*100:.1f}%)")
            logging.info(f"Learning Rate: {current_lr:.2e}")

            # Check for improvement
            is_best = val_iou > self.best_iou
            if is_best:
                self.best_iou = val_iou
                epochs_without_improvement = 0
                logging.info(f"New best IoU: {self.best_iou:.4f}")
            else:
                epochs_without_improvement += 1

            # Save checkpoint every epoch
            self.save_checkpoint(epoch, is_best=is_best)

            # Early stopping check
            if epochs_without_improvement >= early_stopping_patience:
                logging.info(f"Early stopping triggered after {epochs_without_improvement} epochs without improvement")
                break

        total_time = time.time() - start_time
        logging.info(f"\nTraining completed!")
        logging.info(f"Total time: {total_time/3600:.2f} hours")
        logging.info(f"Best IoU achieved: {self.best_iou:.4f} ({self.best_iou*100:.1f}%)")

        # Plot training history
        self.plot_training_history()

        return self.training_history

    def plot_training_history(self):
        if not self.training_history['train_loss']:
            return

        epochs = range(1, len(self.training_history['train_loss']) + 1)

        fig, axes = plt.subplots(2, 2, figsize=(15, 10))

        # Loss plot
        axes[0, 0].plot(epochs, self.training_history['train_loss'], label='Train Loss', color='blue')
        axes[0, 0].plot(epochs, self.training_history['val_loss'], label='Val Loss', color='red')
        axes[0, 0].set_title('Training and Validation Loss')
        axes[0, 0].set_xlabel('Epoch')
        axes[0, 0].set_ylabel('Loss')
        axes[0, 0].legend()
        axes[0, 0].grid(True)

        # IoU plot
        axes[0, 1].plot(epochs, self.training_history['val_iou'], label='Val IoU', color='green')
        axes[0, 1].set_title('Validation IoU')
        axes[0, 1].set_xlabel('Epoch')
        axes[0, 1].set_ylabel('IoU')
        axes[0, 1].legend()
        axes[0, 1].grid(True)

        # Dice plot
        axes[1, 0].plot(epochs, self.training_history['val_dice'], label='Val Dice', color='orange')
        axes[1, 0].set_title('Validation Dice Score')
        axes[1, 0].set_xlabel('Epoch')

import torch
import cv2
import numpy as np
from PIL import Image
import matplotlib.pyplot as plt
import os

class PredictionEngine:
    def __init__(self, model, device, checkpoint_path=None):
        self.model = model
        self.device = device

        if checkpoint_path and os.path.exists(checkpoint_path):
            self.load_model(checkpoint_path)

        self.model.eval()

    def load_model(self, checkpoint_path):
        checkpoint = torch.load(checkpoint_path, map_location=self.device)
        self.model.load_state_dict(checkpoint['model_state_dict'])
        print(f"Model loaded from: {checkpoint_path}")

        if 'best_iou' in checkpoint:
            print(f"Model best IoU: {checkpoint['best_iou']:.4f}")

    def preprocess_image(self, image_path, target_size=320):
        # Load image
        if isinstance(image_path, str):
            image = cv2.imread(image_path)
            image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        else:
            image = image_path

        original_size = image.shape[:2]

        # Resize image
        image_resized = cv2.resize(image, (target_size, target_size))

        # Normalize
        image_normalized = image_resized.astype(np.float32) / 255.0

        # Convert to tensor
        image_tensor = torch.from_numpy(image_normalized).permute(2, 0, 1)

        # Apply normalization
        mean = torch.tensor([0.485, 0.456, 0.406]).view(3, 1, 1)
        std = torch.tensor([0.229, 0.224, 0.225]).view(3, 1, 1)
        image_tensor = (image_tensor - mean) / std

        # Add batch dimension
        image_tensor = image_tensor.unsqueeze(0)

        return image_tensor, original_size, image

    def predict_single(self, image_path, return_prob=False):
        image_tensor, original_size, original_image = self.preprocess_image(image_path)
        image_tensor = image_tensor.to(self.device)

        with torch.no_grad():
            outputs = self.model(image_tensor)

            # Get main output
            if isinstance(outputs, tuple):
                main_output = outputs[0]
            else:
                main_output = outputs

            # Convert to CPU and numpy
            prediction = main_output.squeeze().cpu().numpy()

            # Resize to original size
            prediction_resized = cv2.resize(prediction, (original_size[1], original_size[0]))

            if return_prob:
                return prediction_resized, original_image
            else:
                # Binarize prediction
                binary_mask = (prediction_resized > 0.5).astype(np.uint8) * 255
                return binary_mask, original_image

    def predict_batch(self, image_paths):
        predictions = []

        for image_path in image_paths:
            mask, original = self.predict_single(image_path)
            predictions.append({
                'image_path': image_path,
                'mask': mask,
                'original': original
            })

        return predictions

    def visualize_prediction(self, image_path, save_path=None):
        mask, original = self.predict_single(image_path)

        # Create visualization
        fig, axes = plt.subplots(1, 3, figsize=(15, 5))

        # Original image
        axes[0].imshow(original)
        axes[0].set_title('Original Image')
        axes[0].axis('off')

        # Predicted mask
        axes[1].imshow(mask, cmap='gray')
        axes[1].set_title('Predicted Mask')
        axes[1].axis('off')

        # Overlay
        overlay = original.copy()
        overlay[mask > 127] = [0, 255, 0]  # Green overlay
        blended = cv2.addWeighted(original, 0.7, overlay, 0.3, 0)

        axes[2].imshow(blended)
        axes[2].set_title('Overlay Result')
        axes[2].axis('off')

        plt.tight_layout()

        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"Visualization saved: {save_path}")

        plt.show()

        return mask, original

    def create_passport_photo(self, image_path, background_color='white', output_size=(413, 531)):
        mask, original = self.predict_single(image_path)

        # Create background
        if background_color == 'white':
            bg_color = [255, 255, 255]
        elif background_color == 'blue':
            bg_color = [70, 130, 180]
        elif background_color == 'light_blue':
            bg_color = [173, 216, 230]
        else:
            bg_color = [255, 255, 255]

        # Create background image
        background = np.full(original.shape, bg_color, dtype=np.uint8)

        # Apply mask
        mask_3ch = np.stack([mask, mask, mask], axis=2) / 255.0
        result = (original * mask_3ch + background * (1 - mask_3ch)).astype(np.uint8)

        # Resize to passport dimensions
        result_pil = Image.fromarray(result)
        passport_photo = result_pil.resize(output_size, Image.Resampling.LANCZOS)

        return np.array(passport_photo)

    def evaluate_on_dataset(self, test_image_dir, test_mask_dir):
        # Get test files
        image_files = [f for f in os.listdir(test_image_dir)
                      if f.lower().endswith(('.jpg', '.jpeg', '.png'))]

        total_iou = 0.0
        total_dice = 0.0
        valid_samples = 0

        for img_file in image_files:
            img_path = os.path.join(test_image_dir, img_file)

            # Find corresponding mask
            base_name = os.path.splitext(img_file)[0]
            mask_file = f"{base_name}.png"
            mask_path = os.path.join(test_mask_dir, mask_file)

            if not os.path.exists(mask_path):
                continue

            # Load ground truth mask
            gt_mask = cv2.imread(mask_path, cv2.IMREAD_GRAYSCALE)
            gt_mask = (gt_mask > 127).astype(np.uint8)

            # Get prediction
            pred_mask, _ = self.predict_single(img_path)
            pred_mask = cv2.resize(pred_mask, (gt_mask.shape[1], gt_mask.shape[0]))
            pred_mask = (pred_mask > 127).astype(np.uint8)

            # Calculate IoU
            intersection = np.sum(pred_mask * gt_mask)
            union = np.sum(pred_mask) + np.sum(gt_mask) - intersection
            iou = intersection / (union + 1e-8)

            # Calculate Dice
            dice = (2 * intersection) / (np.sum(pred_mask) + np.sum(gt_mask) + 1e-8)

            total_iou += iou
            total_dice += dice
            valid_samples += 1

        if valid_samples > 0:
            avg_iou = total_iou / valid_samples
            avg_dice = total_dice / valid_samples

            print(f"Evaluation Results:")
            print(f"Average IoU: {avg_iou:.4f} ({avg_iou*100:.1f}%)")
            print(f"Average Dice: {avg_dice:.4f} ({avg_dice*100:.1f}%)")
            print(f"Samples evaluated: {valid_samples}")

            return avg_iou, avg_dice
        else:
            print("No valid samples found for evaluation")
            return 0.0, 0.0

def apply_backgrounds(original_image, mask, user_choice="all"):
    """Apply background replacement"""

    print("Applying backgrounds...")

    all_backgrounds = get_background_colors()

    if user_choice == "passport":
        selected_bg = {k: v for k, v in all_backgrounds.items() if "passport" in k}
    elif user_choice == "visa":
        selected_bg = {k: v for k, v in all_backgrounds.items() if "visa" in k}
    elif user_choice == "document":
        selected_bg = {k: v for k, v in all_backgrounds.items() if "document" in k}
    else:
        selected_bg = all_backgrounds

    results = {}

    # Ensure mask is float32 and in 0-1 range
    if mask.max() > 1:
        mask = mask.astype(np.float32) / 255.0

    mask_3ch = np.stack([mask, mask, mask], axis=-1)

    for bg_key, bg_info in selected_bg.items():

        if bg_info["color"] is None:  # Transparent
            # Create RGBA image for transparent background
            rgba_image = np.zeros((original_image.shape[0], original_image.shape[1], 4), dtype=np.uint8)
            rgba_image[:, :, :3] = original_image
            rgba_image[:, :, 3] = (mask * 255).astype(np.uint8) # Use mask as alpha channel
            results[bg_key] = rgba_image

        else:  # Solid background
            background = np.full_like(original_image, bg_info["color"], dtype=np.uint8)
            # Blend original image (where mask is 1) with background (where mask is 0)
            result = (original_image * mask_3ch + background * (1 - mask_3ch)).astype(np.uint8)
            results[bg_key] = result

        print("Applied", bg_info['name'], "background")

    return results

def model_prediction_step(image_path, model_path):
    """Step 1: Model prediction using your trained model"""

    print("STEP 1: Model Prediction")

    try:
        device = torch.device('cpu')
        checkpoint = torch.load(model_path, map_location=device)
        print("Model checkpoint loaded successfully")

        image = cv2.imread(image_path)
        if image is None:
            print("Could not load image:", image_path)
            return None, None

        image_rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        original_size = image_rgb.shape[:2]

        transform = A.Compose([
            A.Resize(320, 320),
            A.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225]),
            ToTensorV2()
        ])

        transformed = transform(image=image_rgb)
        input_tensor = transformed['image'].unsqueeze(0)

        print("Image preprocessed for model")

        # Temporary dummy mask until model architecture is fixed
        height, width = original_size
        mask = np.zeros((height, width), dtype=np.float32)
        center_x, center_y = width//2, height//2
        cv2.ellipse(mask, (center_x, center_y), (width//3, int(height*0.6)), 0, 0, 360, 1.0, -1)

        print("PREDICTION COMPLETE: Segmentation mask generated")

        return mask, image_rgb

    except Exception as e:
        print("Error in model prediction:", str(e))
        return None, None

def enhance_image_quality(image):
    """Enhance image quality for professional photos"""

    print("STEP 2: Enhancing image quality")

    img_pil = Image.fromarray(image)

    sharpness_enhancer = ImageEnhance.Sharpness(img_pil)
    img_pil = sharpness_enhancer.enhance(1.1)

    contrast_enhancer = ImageEnhance.Contrast(img_pil)

    img_pil = color_enhancer.enhance(1.02)

    enhanced_image = np.array(img_pil)
    print("Image quality enhanced")

    return enhanced_image

def apply_backgrounds(original_image, mask, user_choice="all"):
    """Apply background replacement"""

    print("STEP 3: Applying backgrounds")

    all_backgrounds = get_background_colors()

    if user_choice == "passport":
        selected_bg = {k: v for k, v in all_backgrounds.items() if "passport" in k}
    elif user_choice == "visa":
        selected_bg = {k: v for k, v in all_backgrounds.items() if "visa" in k}
    elif user_choice == "document":
        selected_bg = {k: v for k, v in all_backgrounds.items() if "document" in k}
    else:
        selected_bg = all_backgrounds

    results = {}

    if mask.max() > 1:
        mask = mask.astype(np.float32) / 255.0

    mask_3ch = np.stack([mask, mask, mask], axis=-1)

    for bg_key, bg_info in selected_bg.items():

        if bg_info["color"] is None:  # Transparent
            rgba_image = np.zeros((original_image.shape[0], original_image.shape[1], 4), dtype=np.uint8)
            rgba_image[:, :, :3] = original_image
            rgba_image[:, :, 3] = (mask * 255).astype(np.uint8)
            results[bg_key] = rgba_image

        else:  # Solid background
            background = np.full_like(original_image, bg_info["color"], dtype=np.uint8)
            result = (original_image * mask_3ch + background * (1 - mask_3ch)).astype(np.uint8)
            results[bg_key] = result

        print("Applied", bg_info['name'], "background")

    return results

# STEP 4: PASSPORT DIMENSIONS
def get_passport_dimensions():
    """Define passport dimensions"""

    return {
        "US_passport": {"size": (600, 600), "description": "2x2 inches"},
        "EU_passport": {"size": (413, 531), "description": "35x45mm"},
        "India_passport": {"size": (600, 600), "description": "2x2 inches"},
        "visa_photo": {"size": (600, 600), "description": "2x2 inches"},
        "ID_card": {"size": (400, 500), "description": "ID size"}
    }

def crop_to_passport_size(image, dimension_type="US_passport"):
    """Crop image to passport dimensions"""

    dimensions = get_passport_dimensions()
    target_info = dimensions.get(dimension_type, dimensions["US_passport"])
    target_width, target_height = target_info["size"]
    description = target_info["description"]

    height, width = image.shape[:2]

    crop_ratio = 0.75
    crop_height = int(height * crop_ratio)

    aspect_ratio = target_width / target_height
    crop_width = min(width, int(crop_height * aspect_ratio))

    start_x = max(0, (width - crop_width) // 2)
    start_y = max(0, int(height * 0.1))
    end_x = min(width, start_x + crop_width)
    end_y = min(height, start_y + crop_height)

    cropped = image[start_y:end_y, start_x:end_x]
    resized = cv2.resize(cropped, (target_width, target_height), interpolation=cv2.INTER_LANCZOS4)

    print("Cropped to", target_width, "x", target_height, "(" + description + ")")
    return resized

def save_professional_formats(image, base_filename, output_dir="passport_photos_final"):
    """Save files in professional formats"""

    print("STEP 5: Saving", base_filename)

    os.makedirs(output_dir, exist_ok=True)
    saved_files = []

    if isinstance(image, np.ndarray):
        if len(image.shape) == 3 and image.shape[2] == 4:
            img_pil = Image.fromarray(image, mode='RGBA')
        else:
            img_pil = Image.fromarray(image)
    else:
        img_pil = image

    has_transparency = (img_pil.mode == 'RGBA')

    if has_transparency:
        # Save PNG
        png_path = os.path.join(output_dir, base_filename + ".png")
        img_pil.save(png_path, "PNG", optimize=True, dpi=(300, 300))
        saved_files.append(png_path)

        # Save JPG with white background
        jpg_path = os.path.join(output_dir, base_filename + "_solid_bg.jpg")
        white_bg = Image.new('RGB', img_pil.size, (255, 255, 255))
        white_bg.paste(img_pil, mask=img_pil.split()[-1])
        white_bg.save(jpg_path, "JPEG", quality=95, optimize=True, dpi=(300, 300))
        saved_files.append(jpg_path)

    else:
        # Save JPG
        jpg_path = os.path.join(output_dir, base_filename + ".jpg")
        img_pil.save(jpg_path, "JPEG", quality=95, optimize=True, dpi=(300, 300))
        saved_files.append(jpg_path)

        # Save PNG
        png_path = os.path.join(output_dir, base_filename + ".png")
        img_pil.save(png_path, "PNG", optimize=True, dpi=(300, 300))
        saved_files.append(png_path)

        # Save web JPG
        web_jpg_path = os.path.join(output_dir, base_filename + "_web.jpg")
        img_pil.save(web_jpg_path, "JPEG", quality=85, optimize=True, dpi=(72, 72))
        saved_files.append(web_jpg_path)

    for file_path in saved_files:
        size_kb = os.path.getsize(file_path) / 1024
        print("Saved:", os.path.basename(file_path), "(" + str(int(size_kb)) + " KB)")

    return saved_files

def run_complete_pipeline(image_path, model_path, user_preference="all", base_name="passport"):
    """Run complete passport photo pipeline"""

    print("RUNNING COMPLETE AI PASSPORT PHOTO PIPELINE")
    print("=" * 45)

    # Step 1: Prediction
    mask, original_image = model_prediction_step(image_path, model_path)

    if mask is None or original_image is None:
        print("Pipeline failed at prediction step")
        return None, None

    # Step 2: Enhancement
    enhanced_image = enhance_image_quality(original_image)

    # Step 3: Backgrounds
    background_results = apply_backgrounds(enhanced_image, mask, user_preference)

    # Step 4 & 5: Crop and Save
    dimension_sets = {
        "passport": ["US_passport", "India_passport"],
        "visa": ["visa_photo", "US_passport"],
        "document": ["ID_card"],
        "all": ["US_passport", "EU_passport", "visa_photo", "ID_card"]
    }

    target_dimensions = dimension_sets.get(user_preference, dimension_sets["all"])

    all_files = []

    for bg_key, bg_image in background_results.items():
        for dim_type in target_dimensions:

            print("Processing", bg_key, "-", dim_type)

            # Handle RGBA properly
            if len(bg_image.shape) == 3 and bg_image.shape[2] == 4:
                rgb_part = bg_image[:,:,:3]
                alpha_part = bg_image[:,:,3]

                cropped_rgb = crop_to_passport_size(rgb_part, dim_type)
                cropped_alpha = crop_to_passport_size(alpha_part, dim_type)

                final_cropped = np.dstack([cropped_rgb, cropped_alpha])
            else:
                final_cropped = crop_to_passport_size(bg_image, dim_type)

            filename = base_name + "_" + bg_key + "_" + dim_type
            saved_files = save_professional_formats(final_cropped, filename)
            all_files.extend(saved_files)

            print("Generated", bg_key, "-", dim_type)

    # Step 6: Documentation
    processing_info = {
        "user_preference": user_preference,
        "total_files": len(all_files),
        "backgrounds": len(background_results),
        "dimensions": len(target_dimensions)
    }

    create_documentation(processing_info, "passport_photos_final")

    print("PIPELINE COMPLETE!")
    print("Generated", len(all_files), "professional passport photos")
    print("Output folder: passport_photos_final/")
    print("Ready for app deployment!")

    return all_files, processing_info
